set_1_data <- readRDS(
  "./data/probe_mapping/curated/probe_mapping_data.RDS"
)
set_2_data <- readRDS(
  paste0(
    "./data/probe_mapping/deprecated/",
    "reference_probe_coverage_data_wo6K.RDS"
  )
)

validate_processed_data(set_1_data)
validate_processed_data(set_2_data)

gene_keys <- c("gene_index", "seqid", "start", "end")

set_1_genes <- set_1_data$gene_coverage %>%
  select(
    all_of(gene_keys),
    gene_id,
    locus_tag,
    gene_symbol,
    set_1_covered_bp = covered_bp,
    set_1_coverage_percent = coverage_percent,
    set_1_coverage_status = coverage_status
  )

set_2_genes <- set_2_data$gene_coverage %>%
  select(
    all_of(gene_keys),
    set_2_covered_bp = covered_bp,
    set_2_coverage_percent = coverage_percent,
    set_2_coverage_status = coverage_status
  )

gene_comparison <- set_1_genes %>%
  inner_join(set_2_genes, by = gene_keys) %>%
  mutate(
    set_1_coverage_status = factor(
      set_1_coverage_status,
      levels = coverage_status_levels
    ),
    set_2_coverage_status = factor(
      set_2_coverage_status,
      levels = coverage_status_levels
    ),
    coverage_change = set_1_coverage_percent - set_2_coverage_percent,
    coverage_changed = abs(coverage_change) > sqrt(.Machine$double.eps),
    status_changed = set_1_coverage_status != set_2_coverage_status
  )

if (
  nrow(gene_comparison) != nrow(set_1_genes) ||
    nrow(gene_comparison) != nrow(set_2_genes) ||
    anyDuplicated(gene_comparison[gene_keys]) > 0
) {
  stop("The two probe sets did not match one-to-one across reference genes")
}

status_transitions <- gene_comparison %>%
  count(
    set_1_coverage_status,
    set_2_coverage_status,
    name = "genes",
    .drop = FALSE
  ) %>%
  group_by(set_1_coverage_status) %>%
  mutate(
    display_weight = genes / sum(genes),
    count_weight = if_else(
      display_weight > 0,
      genes / display_weight,
      0
    )
  ) %>%
  ungroup()

p_status_transitions <- ggplot(
  status_transitions,
  aes(
    axis1 = set_1_coverage_status,
    axis2 = set_2_coverage_status,
    y = display_weight,
    weight = count_weight
  )
) +
  geom_alluvium(
    aes(fill = set_1_coverage_status),
    width = 0.18,

coverage_change_summary <- gene_comparison %>%

    genes = n(),

    status_changed = sum(status_changed),
    median_change = median(coverage_change),
