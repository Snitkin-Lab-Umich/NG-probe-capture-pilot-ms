major_sts <- genome_groups %>% subset(st_class == "Major ST") %>% .$ST %>% unique
 

# Merge data
overall_data <- pathogenwatch_isolate_data %>%
  mutate(study = "Overall")
overall_data$major_ST <- overall_data$sequence_type %in% major_sts

chosen_data <- pathogenwatch_isolate_data %>%
  filter(is_chosen) %>%
  mutate(study = "Chosen")
chosen_data$major_ST <-  chosen_data$sequence_type %in% major_sts

comparison_data <- bind_rows(overall_data, chosen_data)

Resistance distribution

resistance_distribution <- comparison_data %>%
  pivot_longer(
    cols = all_of(resistance_variables),
    names_to = "antibiotic",
    values_to = "prediction"
  ) %>%
  group_by(antibiotic, study) %>%
  summarise(
    nonsusceptible = sum(prediction != "NOT_FOUND"),
    total = n(),
    percent = 100 * mean(prediction != "NOT_FOUND"),
    .groups = "drop"
  )

resistance_p_values <- map_dfr(
  resistance_variables,
  function(antibiotic) {
    nonsusceptible <- comparison_data[[antibiotic]] != "NOT_FOUND"
    tibble(
      antibiotic = antibiotic,
      fisher_p = fisher.test(
        table(nonsusceptible, comparison_data$study)
      )$p.value
    )
  }
)

resistance_comparison <- resistance_distribution %>%
  pivot_wider(
    names_from = study,
    values_from = c(nonsusceptible, total, percent)
  ) %>%
  left_join(resistance_p_values, by = "antibiotic") %>%
  arrange(match(antibiotic, resistance_variables))

mutation_distribution <- comparison_data %>%
  pivot_longer(
    cols = all_of(mutation_variables),
    names_to = "locus",
    values_to = "value"
  ) %>%
  mutate(present = as.numeric(as.character(value)) > 0) %>%
  group_by(locus, study) %>%
  summarise(
    present = sum(present),
    total = n(),
    percent = 100 * mean(present),
    .groups = "drop"
  )

mutation_p_values <- map_dfr(
  mutation_variables,
  function(locus) {
    present <- as.numeric(as.character(comparison_data[[locus]])) > 0
    tibble(
      locus = locus,
      fisher_p = fisher.test(
        table(present, comparison_data$study)
      )$p.value
    )
  }
)

mutation_comparison <- mutation_distribution %>%
