# Chosen assemblies capture genotype distributions despite shifting lineage compositions

overall_data <- pathogenwatch_isolate_data %>%
  mutate(study = "Overall")

chosen_data <- pathogenwatch_isolate_data %>%
  filter(is_chosen) %>%
  mutate(study = "Chosen")

comparison_data <- bind_rows(overall_data, chosen_data)

temporal_comparison <- comparison_data %>%
  filter(!is.na(collection_decade)) %>%
  group_by(study) %>%
  summarise(
    genomes_with_year = n(),
    genomes_in_2010s = sum(collection_decade == 2010),
    percent_in_2010s = 100 * mean(collection_decade == 2010),
    .groups = "drop"
  )

set.seed(45)
decade_fisher_p <- fisher.test(
  table(comparison_data$collection_decade, comparison_data$study),
  simulate.p.value = TRUE
)$p.value

continent_comparison <- comparison_data %>%
  filter(!is.na(collection_continent)) %>%
  count(study, collection_continent) %>%
  group_by(study) %>%
  mutate(
    total = sum(n),
    percent = 100 * n / total
  ) %>%
  ungroup()

sequence_type_comparison <- tibble(
  global_sequence_types = n_distinct(overall_data$sequence_type, na.rm = TRUE),
  chosen_sequence_types = n_distinct(chosen_data$sequence_type, na.rm = TRUE)
) %>%
  mutate(percent_represented = 100 * chosen_sequence_types / global_sequence_types)

set.seed(45)
sequence_type_fisher_p <- fisher.test(
  table(comparison_data$sequence_type, comparison_data$study),
  simulate.p.value = TRUE
)$p.value

resistance_distribution <- comparison_data %>%
  pivot_longer(
    cols = all_of(resistance_variables),
    names_to = "antibiotic",
    values_to = "prediction"
  ) %>%
  group_by(antibiotic, study) %>%
  summarise(
    nonsusceptible = sum(prediction != "NOT_FOUND"),
    total = n(),
    percent = 100 * mean(prediction != "NOT_FOUND"),
    .groups = "drop"
  )

resistance_p_values <- map_dfr(
  resistance_variables,
  function(antibiotic) {
    nonsusceptible <- comparison_data[[antibiotic]] != "NOT_FOUND"
    tibble(
      antibiotic = antibiotic,
      fisher_p = fisher.test(
        table(nonsusceptible, comparison_data$study)
      )$p.value
    )
  }
)

resistance_comparison <- resistance_distribution %>%
  pivot_wider(
    names_from = study,
    values_from = c(nonsusceptible, total, percent)
  ) %>%
  left_join(resistance_p_values, by = "antibiotic") %>%
  arrange(match(antibiotic, resistance_variables))

mutation_distribution <- comparison_data %>%
  pivot_longer(
    cols = all_of(mutation_variables),
    names_to = "locus",
    values_to = "value"
  ) %>%
  mutate(present = as.numeric(as.character(value)) > 0) %>%
  group_by(locus, study) %>%
  summarise(
    present = sum(present),
    total = n(),
    percent = 100 * mean(present),
    .groups = "drop"
  )

mutation_p_values <- map_dfr(
  mutation_variables,
  function(locus) {
    present <- as.numeric(as.character(comparison_data[[locus]])) > 0
    tibble(
      locus = locus,
      fisher_p = fisher.test(
        table(present, comparison_data$study)
      )$p.value
    )
  }
)

mutation_comparison <- mutation_distribution %>%
