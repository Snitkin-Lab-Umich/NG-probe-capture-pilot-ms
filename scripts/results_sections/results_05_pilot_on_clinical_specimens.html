add_contig_barrier <-  function(data) {
  data %>%
    mutate(
      dist_from_start = contig_start - 1,
      dist_from_end = length - contig_stop,
      min_dist_to_edge = pmin(dist_from_start, dist_from_end)
    )
}

Coverage data

locus_position_coverage <- coverage_qc %>%
  group_by(locus, locus_position) %>%
  summarise(
    isolates_with_coverage = sum(depth > 0),
    .groups = "drop"
  )

locus_coverage_summary <- locus_position_coverage %>%
  summarise(
    loci = n_distinct(locus),
    locus_positions = n(),
    positions_with_coverage = sum(isolates_with_coverage > 0),
    positions_with_coverage_percent = 100 * mean(isolates_with_coverage > 0)
  )

mutation_position_coverage <- mutation_positions %>%
  left_join(
    locus_position_coverage,
    by = c("locus", "locus_position")
  )

mutation_isolate_coverage <- coverage_qc %>%
  inner_join(
    mutation_positions,
    by = c("locus", "locus_position"),
    relationship = "many-to-one"
  )

mutation_coverage_summary <- mutation_position_coverage %>%
  summarise(
    database_positions = n(),
    positions_covered_in_at_least_one_isolate = sum(isolates_with_coverage > 0),
    positions_covered_in_all_isolates = sum(isolates_with_coverage == length(isolates))
  ) %>%
  mutate(
    isolate_position_observations = nrow(mutation_isolate_coverage),
    isolate_position_observations_with_coverage = sum(
      mutation_isolate_coverage$depth > 0
    ),
    isolate_position_observations_with_coverage_percent =
      100 * mean(mutation_isolate_coverage$depth > 0)
  )

called_locus_coverage_20x <- coverage_breadth_data %>%
  filter(depth_threshold == 20) %>%
  semi_join(gene_calls, by = c("isolate", "locus")) %>%
  summarise(
    isolate_locus_pairs_at_least_90_percent = sum(percent_coverage >= 90),
    called_isolate_locus_pairs = n(),
    percent_at_least_90_percent = 100 * mean(percent_coverage >= 90)
  )
