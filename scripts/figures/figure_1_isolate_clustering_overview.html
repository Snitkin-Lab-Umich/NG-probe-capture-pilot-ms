phenotype_vars <- names(profiles_350)[1:8]
genotype_vars <- names(profiles_350)[9:30]

phenotype_change <- map_dfr(
  phenotype_vars,
  function(variable) {
    present <- profiles_350[[variable]] == "RESISTANT"
    tibble(
      trait = paste(variable, "resistant"),
      trait_type = "Phenotype",
      original_proportion = weighted.mean(present, profiles_350$count),
      selected_proportion = mean(present)
    )
  }
)

genotype_change <- map_dfr(
  genotype_vars,
  function(variable) {
    present <- as.integer(as.character(profiles_350[[variable]]))
    tibble(
      trait = str_replace_all(variable, "_", " "),
      trait_type = "Genotype",
      original_proportion = weighted.mean(present, profiles_350$count),
      selected_proportion = mean(present)
    )
  }
) %>% mutate(trait = paste0("<i>",trait,"</i>") %>% gsub("<i>blaTEM</i>","<i>bla</i><sub>TEM</sub>",.))

trait_representation <- bind_rows(phenotype_change, genotype_change) %>%
  mutate(
    change_percentage_points = 100 * (selected_proportion - original_proportion),
    trait_type = factor(trait_type, levels = c("Phenotype", "Genotype")),

# ranking STs in size range for example that shows multiple 350-AA clusters being partitioned into both singleton and multi-genome profiles of different sizes
find_profile_example_sts <- function(
  data,
  min_genomes = 50,
  max_genomes = 60
) {
  data %>%
    filter(!is.na(ST)) %>%
    count(ST, aa_cluster, profile, name = "profile_size") %>%
    group_by(ST) %>%
    summarise(
      n_genomes = sum(profile_size),
      n_clusters = n_distinct(aa_cluster),
      n_profiles = n(),
      singleton_profiles = sum(profile_size == 1),
      multi_genome_profiles = sum(profile_size > 1),
      split_clusters = sum(table(aa_cluster) > 1),
      distinct_profile_sizes = n_distinct(profile_size),
      max_profile_size = max(profile_size),
      .groups = "drop"
    ) %>%
    filter(
      between(n_genomes, min_genomes, max_genomes),
      singleton_profiles > 0,
      multi_genome_profiles > 0,
      split_clusters > 0
    ) %>%
    arrange(
      desc(split_clusters),
      desc(distinct_profile_sizes),
      desc(n_profiles),
      abs(n_genomes - mean(c(min_genomes, max_genomes))),
      ST
    )
}
