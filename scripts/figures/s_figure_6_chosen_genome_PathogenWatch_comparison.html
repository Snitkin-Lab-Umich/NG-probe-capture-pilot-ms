# decade

decade_overlay <- ng_overlay_meta %>%
  filter(!is.na(collection_decade)) %>%
  count(collection_decade, study) %>%
  group_by(study) %>%
  mutate(perc = 100 * n / sum(n)) %>%
  ungroup()

decade_pvals <- lapply(sort(unique(ng_overlay_meta$collection_decade[!is.na(ng_overlay_meta$collection_decade)])), function(d) {
  tmp <- ng_overlay_meta %>% filter(!is.na(collection_decade))
  in_group <- tmp$collection_decade == d
  tbl <- table(
    factor(in_group, levels = c(FALSE, TRUE)),
    factor(tmp$study, levels = c("Overall", "Chosen"))
  )
  data.frame(collection_decade = d, p_value = fisher.test(tbl)$p.value)
}) %>% dplyr::bind_rows()

decade_overlay <- left_join(decade_overlay, decade_pvals, by = "collection_decade")

A_overlay <- ggplot(decade_overlay,
                    aes(x = factor(collection_decade),
                        y = perc,
                        fill = study)) +
  geom_col(position = position_dodge(0.9)) +
  geom_text(
    data = decade_overlay %>%
      group_by(collection_decade) %>%
      slice_max(perc, n = 1) %>%
      filter(p_value < 0.05),
    aes(label = "*"),
    position = position_dodge(0.9),
    vjust = -0.6,
    size = 5
  ) +
  study_fill_scale +
  theme_bw_me +
  xlab("Decade") +
  ylab("Percentage (%)")

res_overlay <- ng_overlay_meta %>%
  select(genome_id, study, all_of(resistance_variables)) %>%

               names_to = "Antibiotic",

  mutate(binary = Status != "NOT_FOUND") %>%
  group_by(Antibiotic, study) %>%
  summarise(perc = 100 * mean(binary),

# genotype
mut_overlay <- ng_overlay_meta %>%
  select(genome_id, study, all_of(mutation_variables)) %>%
  pivot_longer(cols = all_of(mutation_variables),
               names_to = "Genotype",
               values_to = "Presence") %>%
  mutate(Presence = as.numeric(as.character(Presence))) %>%
  group_by(Genotype, study) %>%
  summarise(perc = 100 * mean(Presence),
            .groups = "drop")

mut_pvals <- lapply(mutation_variables, function(x) {
  tmp <- ng_overlay_meta %>%
    mutate(binary = as.numeric(.data[[x]]) > 0)
  tbl <- table(
    factor(tmp$binary, levels = c(FALSE, TRUE)),
    factor(tmp$study, levels = c("Overall", "Chosen"))
  )
  data.frame(
    Genotype = x,
    p_value = fisher.test(tbl)$p.value
  )
}) %>% dplyr::bind_rows()
 
## derive locus from genotype name
mut_overlay_locus <- mut_overlay %>%
  mutate(Locus = sub("_.*", "", Genotype)) %>%
  group_by(Locus, study) %>%
  summarise(
    perc = mean(perc),
    .groups = "drop"
  )

mut_pvals_locus <- lapply(unique(mut_overlay_locus$Locus), function(lc){

  tmp <- ng_overlay_meta %>%
    select(genome_id, study, all_of(mutation_variables)) %>%
    pivot_longer(cols = all_of(mutation_variables),
                 names_to = "Genotype",
                 values_to = "Presence") %>%
    mutate(
      Locus = sub("_.*", "", Genotype),
      Presence = as.numeric(as.character(Presence))
    ) %>%
    group_by(genome_id, study, Locus) %>%
    summarise(binary = any(Presence > 0),
              .groups = "drop") %>%
    filter(Locus == lc)

  tbl <- table(
    factor(tmp$binary, levels = c(FALSE, TRUE)),
    factor(tmp$study, levels = c("Overall", "Chosen"))
  )

  data.frame(
    Locus = lc,
    p_value = fisher.test(tbl)$p.value
  )

}) %>% dplyr::bind_rows()

mut_overlay_locus <- left_join(mut_overlay_locus,
                               mut_pvals_locus,
                               by = "Locus")

mut_overlay_locus$Locus <-  paste0("<i>",mut_overlay_locus$Locus,"</i>") %>% gsub("<i>blaTEM</i>","<i>bla</i><sub>TEM</sub>",.)

locus_order <- mut_overlay_locus %>%
  filter(study == "Overall") %>%
  arrange(desc(perc)) %>%
  pull(Locus)

mut_overlay_locus$Locus <- factor(mut_overlay_locus$Locus, levels = locus_order)

E_overlay <- ggplot(mut_overlay_locus,
