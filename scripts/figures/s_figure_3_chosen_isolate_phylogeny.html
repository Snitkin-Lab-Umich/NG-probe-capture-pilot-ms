col_width_adj <- 0.1
offset_step <- 0.0040625 * col_width_adj

  select(genome_id, st_track) %>%
  column_to_rownames("genome_id") %>%
  `colnames<-`("Major STs")

  method = "binary"

  .$order

  p3_350.1,
  mutation_data_350 %>%

add legend

get_legend <- function(p) cowplot::get_legend(p + theme(legend.position = "bottom"))

legend_st_plot <- ggplot(
  data.frame(
    st_track = factor(
      c(major_sts, "Other ST"),
      levels = c(major_sts, "Other ST")
    )
  ),
  aes(x = 1, y = st_track, fill = st_track)
) +
  geom_tile() +
  st_scale_fill +
  theme_void() +
  theme(
    legend.position = "bottom",
    legend.box = "horizontal",
    legend.title.position = "top",
    legend.text = element_text(size = 10),
    legend.title = element_text(size = 12),
    legend.key.size = unit(0.3, "cm"),legend.key = element_rect(colour = c('black'))
  ) +
  guides(
    fill = guide_legend(
      ncol = 10,
      byrow = TRUE,
      title.position = "top",
      label.position = "right"
    )
  )

legend_res_plot <- ggplot(
  data.frame(Resistance = factor(resistance_levels, levels = resistance_levels)),
  aes(x = 1, y = Resistance, fill = Resistance)
) +
  geom_tile() +

  theme_void() +
  theme(
    legend.position = "bottom",
    legend.box = "horizontal",
