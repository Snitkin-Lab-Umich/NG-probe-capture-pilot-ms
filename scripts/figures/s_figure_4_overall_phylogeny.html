# Center the largest contiguous run of minor/Other-ST tips at 12 o'clock,
# creating a gap in the annotation rings for their numbered labels.
largest_circular_run <- function(x) {
  n <- length(x)
  if (all(x)) return(seq_len(n))
  runs <- rle(c(x, x))
  run_ends <- cumsum(runs$lengths)
  run_starts <- run_ends - runs$lengths + 1
  candidates <- which(runs$values & run_starts <= n)
  best <- candidates[which.max(runs$lengths[candidates])]
  ((seq.int(run_starts[best], length.out = runs$lengths[best]) - 1) %% n) + 1
}

circular_tip_order <- circular_tree$data %>%
  filter(isTip) %>%
  transmute(genome_id = label, y) %>%
  left_join(
    phylogeny_track_data %>% select(genome_id, st_track),
    by = "genome_id"
  ) %>%
  arrange(y)

minor_gap_indices <- largest_circular_run(
  circular_tip_order$st_track == "Other ST"
)
minor_gap_center <- (
  minor_gap_indices[1] - 1 + (length(minor_gap_indices) - 1) / 2
) %% nrow(circular_tip_order) + 1
minor_gap_rotation <- 360 * (minor_gap_center - 0.5) /
  nrow(circular_tip_order)

circular_tree <- gheatmap(
  circular_tree,
  phylogeny_tracks,
  width = 0.12,
  offset = 0,
  color = NA,
  colnames = FALSE
) +
  scale_fill_manual(
    values = track_colors,
    breaks = c(major_sts, "Other ST")
  ) +
  theme(
    legend.position = "none",
    plot.margin = margin(0, 0, 0, 0)
  )

circular_tree <- ggtree::rotate_tree(
  circular_tree,
  angle = minor_gap_rotation
)
