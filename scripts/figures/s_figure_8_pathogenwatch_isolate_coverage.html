isolate_matrices <- setNames(lapply(isolates, function(sample_name) {
    sample_data <- filter(coverage_qc, isolate == sample_name)
    mat <- matrix(
        NA_real_,
        length(ordered_loci),
        length(locus_positions),
        dimnames = list(ordered_loci, locus_positions)
    )
    indices <- cbind(
        match(sample_data$locus, ordered_loci),
        sample_data$locus_position
    )
    mat[indices] <- sample_data$normalized_depth
    stopifnot(identical(unname(mat[indices]), sample_data$normalized_depth),
              sum(!is.na(mat)) == nrow(sample_data))
    mat
}), isolates)

make_panel <- function(sample_name) {
    mat <- isolate_matrices[[sample_name]]
    called_loci <- gene_calls$locus[gene_calls$isolate == sample_name]
    panel_locus_labels <- paste0(
        gsub("_", " ", rrna_display(ordered_loci)),
        ifelse(ordered_loci %in% called_loci, " (called)", "")
    )
    ht <- Heatmap(
        mat, name = "position_depth", col = coverage_colors, na_col = "white",
        cluster_rows = FALSE, cluster_columns = FALSE,
        row_names_side = "left", row_labels = panel_locus_labels,
        row_names_gp = gpar(fontsize = 9.5),
        column_labels = ifelse(
            locus_positions == 1 | locus_positions %% 1000 == 0,
            locus_positions,
            ""
        ),
        column_names_rot = 0, column_names_gp = gpar(fontsize = 10),
        column_title = "Position (bp)", column_title_side = "bottom",
        column_title_gp = gpar(fontsize = 12),
        show_heatmap_legend = FALSE,
        use_raster = TRUE, raster_device = "agg_png", raster_by_magick = FALSE,
        layer_fun = function(j, i, x, y, width, height, fill) {
            first <- j == 1
            grid.rect(x = unit(locus_lengths[i[first]] / ncol(mat) / 2, "npc"),
                      y = y[first], width = unit(locus_lengths[i[first]] / ncol(mat), "npc"),
                      height = height[first], gp = gpar(fill = NA, col = "black", lwd = 0.6))
            marked <- markers[cbind(i, j)]
            grid.points(x[marked], y[marked], pch = 1, size = unit(1.1, "mm"),
                        gp = gpar(col = "black", lwd = 0.6))
        }
    )
    grid.grabExpr(draw(ht, column_title = isolate_info$panel_title[match(sample_name, isolate_info$isolate)],
                       column_title_gp = gpar(fontsize = 12, fontface = "bold"),
                       padding = unit(c(3, 2, 3, 2), "mm")), width = 5.6, height = 5.3)
}

Each panel shows one QC-passing isolate and its pool, with all 35 Pathogenwatch reference genes retained. Panels are ordered by pool concentration (High, then Medium, then Low). A gene is marked (called) only when AMRSearch reports a gene call for that isolate and reference; unmarked genes have no gene call. Each isolate is normalized by its own chromosome median-nonzero depth, matching the isolate-level normalization used by Figure 4. The common legend shows the Figure 4D color scale: white at zero, blue below 1, yellow at 1, and orange to red up to 2+, with space outside the gene blank. Circles mark database mutation positions, regardless of whether the isolate has the mutation; variants sharing a position share a circle. The four disrupted entries listed above have no defined position.
