B. country

chosen_genomes_meta <- chosen_genomes_meta %>%
  mutate(
    country_group = ifelse(
      collection_country %in% names(which(table(collection_country) < 10)),
      "Other countries",
      collection_country
    )
  )

B <- ggplot(
    data = chosen_genomes_meta,
    aes(x = forcats::fct_infreq(country_group))
) +
    geom_bar(fill = "black") +
    theme_bw_me_large +
    xlab(NULL) +
    ylab("Isolates")  +
    theme(axis.text.x = element_text(size = 9.5, angle = 45, hjust = 1)) + geom_text(
        aes(label = after_stat(count), y = after_stat(count)),
        size = 4.5,
        stat = "count",
        vjust =-0.15,hjust=0,
        angle=45
    ) + ylim(NA,ceiling(max(table(chosen_genomes_meta$country_group))/1000)*1000)

E. mutational data

reshaped_mutation_gene_collapsed <- summarise_mutation_genes(
  chosen_genomes_meta,
  isolate_column = "genome_id",
  mutation_columns = mutation_variables
)


reshaped_mutation_gene_collapsed$gene_rename <- ifelse(
    reshaped_mutation_gene_collapsed$gene == "blaTEM",
    "<i>bla</i><sub>TEM</sub>",
    paste0("<i>", reshaped_mutation_gene_collapsed$gene, "</i>")
)

reshaped_mutation_gene_collapsed$gene_rename <- factor(
reshaped_mutation_gene_collapsed$gene_rename ,levels = reshaped_mutation_gene_collapsed %>% arrange(-n) %>% .$gene_rename)

E <- ggplot(reshaped_mutation_gene_collapsed, aes(x = gene_rename, y = perc)) +
  geom_col(fill = "black") +
  theme_bw_me_large +
  xlab("Gene") +
  ylab("Percent with genotype (%)") +
  geom_text(aes(label = n), size = 3.5, nudge_y = 4, angle = 45) +
